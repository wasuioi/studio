# Firebase → Supabase Migration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Firebase data layer (Auth, Firestore, Storage) with Supabase (Auth, Postgres, Storage) on a single feature branch, preserving every existing feature — email/password auth, real-time messages, real-time chat requests, profile posts, Vera bot, identity verification, and matching.

**Architecture:** Pure `@supabase/supabase-js` + `@supabase/ssr`, no ORM. Seven Postgres tables in the `public` schema with RLS on every table. Profile row is auto-created by a Postgres trigger on `auth.users` insert. Real-time via Supabase Realtime channels that replace Firestore `onSnapshot`. Storage used for profile/post photos only; verification ID photos never persist.

**Tech Stack:** Next.js 15 App Router, React 19, TypeScript, `@supabase/supabase-js`, `@supabase/ssr`, Postgres (via Supabase), Supabase Storage, Supabase Realtime, shadcn/ui, Tailwind, Genkit + Google Gemini (unchanged), Stripe (unchanged), Upstash Redis (unchanged), Resend (kept available but unused in v0).

**Working directory:** `/Users/prawarnaru/TrueVibe/studio` — all commands below assume this is `pwd`.

```sql
-- 001_initial_schema.sql
-- TrueVibe Postgres schema: profiles, posts, personality_profiles, bot_chats,
-- chat_requests, conversations, messages.
-- Replaces: Firestore collections users, users/*/posts, users/*/personality,
-- bot_chats, chat_requests, conversations, conversations/*/messages.

-- =========================================================================
-- Extensions
-- =========================================================================
create extension if not exists "pgcrypto";  -- for gen_random_uuid()

-- =========================================================================
-- profiles — one row per authenticated user
-- =========================================================================
create table public.profiles (
  id                    uuid primary key references auth.users(id) on delete cascade,
  email                 text not null,
  display_name          text,
  bio                   text,
  photo_url             text,
  identity_verified     boolean not null default false,
  verification_status   text not null default 'not_verified'
                          check (verification_status in ('not_verified','pending','verified','rejected')),
  profile_completed     boolean not null default false,
  tier                  text not null default 'free'
                          check (tier in ('free','premium')),
  last_login            timestamptz,
  created_at            timestamptz not null default now(),
  updated_at            timestamptz not null default now()
);

comment on table public.profiles is
  'Replaces Firestore users/{userId}. id is the same UUID as auth.users.id.';

-- =========================================================================
-- posts — user-authored posts shown on profile
-- =========================================================================
create table public.posts (
  id          uuid primary key default gen_random_uuid(),
  user_id     uuid not null references public.profiles(id) on delete cascade,
  content     text,
  photo_url   text,
  created_at  timestamptz not null default now()
);

create index posts_user_id_created_at_idx
  on public.posts (user_id, created_at desc);

comment on table public.posts is
  'Replaces Firestore users/{uid}/posts/*';

-- =========================================================================
-- personality_profiles — finalized Vera interview output (one row per user)
-- =========================================================================
create table public.personality_profiles (
  user_id        uuid primary key references public.profiles(id) on delete cascade,
  partner_vision text,
  traits         jsonb,
  summary        text,
  completed_at   timestamptz
);

comment on table public.personality_profiles is
  'Replaces Firestore users/{uid}/personality/*. Finalized interview output.';

-- =========================================================================
-- bot_chats — Vera 3-phase interview conversation history
-- =========================================================================
create table public.bot_chats (
  id          uuid primary key default gen_random_uuid(),
  user_id     uuid not null references public.profiles(id) on delete cascade,
  phase       int not null default 1,
  messages    jsonb not null default '[]'::jsonb,
  completed   boolean not null default false,
  created_at  timestamptz not null default now(),
  updated_at  timestamptz not null default now()
);

create index bot_chats_user_id_updated_at_idx
  on public.bot_chats (user_id, updated_at desc);

comment on table public.bot_chats is
  'Replaces Firestore bot_chats/{chatId}. Vera interview transcript.';

-- =========================================================================
-- chat_requests — "Can we chat?" requests between users
-- =========================================================================
create table public.chat_requests (
  id          uuid primary key default gen_random_uuid(),
  sender_id   uuid not null references public.profiles(id) on delete cascade,
  receiver_id uuid not null references public.profiles(id) on delete cascade,
  status      text not null default 'pending'
                check (status in ('pending','accepted','rejected')),
  message     text,
  created_at  timestamptz not null default now(),
  updated_at  timestamptz not null default now(),
  unique (sender_id, receiver_id),
  check (sender_id <> receiver_id)
);

create index chat_requests_receiver_status_idx
  on public.chat_requests (receiver_id, status);
create index chat_requests_sender_status_idx
  on public.chat_requests (sender_id, status);

comment on table public.chat_requests is
  'Replaces Firestore chat_requests/{requestId}.';

-- =========================================================================
-- conversations — 1:1 DM threads between two users
-- =========================================================================
create table public.conversations (
  id               uuid primary key default gen_random_uuid(),
  participant_ids  uuid[] not null,
  last_message_at  timestamptz,
  created_at       timestamptz not null default now(),
  check (array_length(participant_ids, 1) = 2),
  check (participant_ids[1] < participant_ids[2])  -- enforce sorted order for uniqueness
);

-- Uniqueness: one conversation per sorted pair of participants
create unique index conversations_participants_uniq
  on public.conversations (participant_ids);

-- GIN index so "user X is in conversation" queries are fast
create index conversations_participants_gin
  on public.conversations using gin (participant_ids);

comment on table public.conversations is
  'Replaces Firestore conversations/{conversationId}. participant_ids must be sorted ascending to prevent duplicate pairs.';

-- =========================================================================
-- messages — individual messages within a conversation
-- =========================================================================
create table public.messages (
  id              uuid primary key default gen_random_uuid(),
  conversation_id uuid not null references public.conversations(id) on delete cascade,
  sender_id       uuid not null references public.profiles(id) on delete cascade,
  content         text not null,
  read_at         timestamptz,
  created_at      timestamptz not null default now()
);

create index messages_conversation_created_at_idx
  on public.messages (conversation_id, created_at desc);

comment on table public.messages is
  'Replaces Firestore conversations/{cid}/messages/*. Immutable — no UPDATE/DELETE policies.';

-- =========================================================================
-- updated_at auto-update trigger (shared)
-- =========================================================================
create or replace function public.set_updated_at()
returns trigger language plpgsql as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

create trigger profiles_set_updated_at
  before update on public.profiles
  for each row execute function public.set_updated_at();

create trigger bot_chats_set_updated_at
  before update on public.bot_chats
  for each row execute function public.set_updated_at();

create trigger chat_requests_set_updated_at
  before update on public.chat_requests
  for each row execute function public.set_updated_at();
```

```sql
-- 002_rls_policies.sql
-- Row Level Security policies replacing firestore.rules.
-- Defaults to DENY — every policy is an explicit allow.

-- =========================================================================
-- Helper functions
-- =========================================================================
create or replace function public.is_email_verified()
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select (auth.jwt()->>'email')::text is not null
     and coalesce((auth.jwt()->>'email_confirmed_at')::text, '') <> '';
$$;

create or replace function public.is_conversation_participant(conv_id uuid)
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select exists (
    select 1 from public.conversations
    where id = conv_id and auth.uid() = any(participant_ids)
  );
$$;

-- =========================================================================
-- profiles
-- =========================================================================
alter table public.profiles enable row level security;

create policy "profiles_select_authenticated"
  on public.profiles for select
  to authenticated
  using (true);

create policy "profiles_update_self"
  on public.profiles for update
  to authenticated
  using (id = auth.uid())
  with check (id = auth.uid());

-- No INSERT policy: only the handle_new_user trigger (SECURITY DEFINER) may insert.
-- No DELETE policy: deletion cascades from auth.users.

-- =========================================================================
-- posts
-- =========================================================================
alter table public.posts enable row level security;

create policy "posts_select_authenticated"
  on public.posts for select
  to authenticated
  using (true);

create policy "posts_insert_owner_verified"
  on public.posts for insert
  to authenticated
  with check (user_id = auth.uid() and public.is_email_verified());

create policy "posts_update_owner"
  on public.posts for update
  to authenticated
  using (user_id = auth.uid())
  with check (user_id = auth.uid());

create policy "posts_delete_owner"
  on public.posts for delete
  to authenticated
  using (user_id = auth.uid());

-- =========================================================================
-- personality_profiles
-- =========================================================================
alter table public.personality_profiles enable row level security;

create policy "personality_profiles_select_owner"
  on public.personality_profiles for select
  to authenticated
  using (user_id = auth.uid());

create policy "personality_profiles_insert_owner"
  on public.personality_profiles for insert
  to authenticated
  with check (user_id = auth.uid());

create policy "personality_profiles_update_owner"
  on public.personality_profiles for update
  to authenticated
  using (user_id = auth.uid())
  with check (user_id = auth.uid());

create policy "personality_profiles_delete_owner"
  on public.personality_profiles for delete
  to authenticated
  using (user_id = auth.uid());

-- =========================================================================
-- bot_chats
-- =========================================================================
alter table public.bot_chats enable row level security;

create policy "bot_chats_select_owner"
  on public.bot_chats for select
  to authenticated
  using (user_id = auth.uid());

create policy "bot_chats_insert_owner"
  on public.bot_chats for insert
  to authenticated
  with check (user_id = auth.uid());

create policy "bot_chats_update_owner"
  on public.bot_chats for update
  to authenticated
  using (user_id = auth.uid())
  with check (user_id = auth.uid());

create policy "bot_chats_delete_owner"
  on public.bot_chats for delete
  to authenticated
  using (user_id = auth.uid());

-- =========================================================================
-- chat_requests
-- =========================================================================
alter table public.chat_requests enable row level security;

create policy "chat_requests_select_participant"
  on public.chat_requests for select
  to authenticated
  using (sender_id = auth.uid() or receiver_id = auth.uid());

create policy "chat_requests_insert_sender_verified"
  on public.chat_requests for insert
  to authenticated
  with check (sender_id = auth.uid() and public.is_email_verified());

create policy "chat_requests_update_participant"
  on public.chat_requests for update
  to authenticated
  using (sender_id = auth.uid() or receiver_id = auth.uid())

  on public.chat_requests for delete
  to authenticated
  using (sender_id = auth.uid() or receiver_id = auth.uid());

-- =========================================================================
-- conversations
-- =========================================================================
alter table public.conversations enable row level security;

create policy "conversations_select_participant_verified"
  on public.conversations for select
  to authenticated
  using (auth.uid() = any(participant_ids) and public.is_email_verified());

create policy "conversations_insert_participant_verified"
  on public.conversations for insert
  to authenticated
  with check (auth.uid() = any(participant_ids) and public.is_email_verified());

create policy "conversations_update_participant_verified"
  on public.conversations for update
  to authenticated
  using (auth.uid() = any(participant_ids) and public.is_email_verified())
  with check (auth.uid() = any(participant_ids) and public.is_email_verified());

create policy "conversations_delete_participant_verified"
  on public.conversations for delete
  to authenticated
  using (auth.uid() = any(participant_ids) and public.is_email_verified());

-- =========================================================================

create policy "messages_insert_sender_participant_verified"

    sender_id = auth.uid()
    and public.is_email_verified()
    and public.is_conversation_participant(conversation_id)
  );

-- No UPDATE or DELETE policies: messages are immutable once sent.
```

```sql
-- Supabase Storage buckets and their RLS policies.
-- Note: bucket creation uses storage.buckets table; files live in storage.objects.
--
-- Two public-read buckets:
```

```tsx
import type { User } from '@supabase/supabase-js';
import { supabase } from '@/supabase/client';
import { SupabaseErrorListener } from '@/supabase/error-listener';

/**
 * React context that exposes the current authenticated Supabase user and
 * a loading flag. Mirrors the shape of the old FirebaseProvider so feature
 * pages can migrate with minimal changes.
 */

interface SupabaseAuthState {
  user: User | null;
  isUserLoading: boolean;
  userError: Error | null;
}

const SupabaseContext = createContext<SupabaseAuthState | undefined>(undefined);

export function SupabaseProvider({ children }: { children: ReactNode }) {
  const [state, setState] = useState<SupabaseAuthState>({
    user: null,
    isUserLoading: true,
    userError: null,
  });

  useEffect(() => {
    let mounted = true;

    // 1. Initial session fetch
    supabase.auth.getSession().then(({ data, error }) => {
      if (!mounted) return;
      setState({
        user: data.session?.user ?? null,
        isUserLoading: false,
        userError: error ? new Error(error.message) : null,
      });
    });

    // 2. Subscribe to subsequent auth changes (login, logout, token refresh)
    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange((_event, session) => {
      if (!mounted) return;
      setState({
        user: session?.user ?? null,
        isUserLoading: false,
        userError: null,
      });
    });

    return () => {
      mounted = false;
      subscription.unsubscribe();
    };
  }, []);

  return (
    <SupabaseContext.Provider value={state}>
      <SupabaseErrorListener />
      {children}
    </SupabaseContext.Provider>
  );
}

  if (ctx === undefined) {
    throw new Error('useSupabaseAuth must be used within SupabaseProvider');
```

```ts
'use client';

import { useEffect, useState } from 'react';
import { supabase } from '@/supabase/client';
import type { Database } from '@/lib/database.types';

type TableName = keyof Database['public']['Tables'];
type Row<T extends TableName> = Database['public']['Tables'][T]['Row'];

interface UseRowResult<T> {
  data: T | null;
  isLoading: boolean;
  error: Error | null;
}

interface UseRowOptions<T extends TableName> {
  table: T;
  column?: string;  // default 'id'
  value: string | number | null | undefined;
  realtime?: boolean; // default false
}

/**
 * React hook to fetch and optionally subscribe to a single row by column+value.
 * Drop-in replacement for Firebase useDoc().
 *
 * Usage:
 *   const { data: profile } = useRow<'profiles'>({
 *     table: 'profiles',
 *     value: userId,
 *   });
 *
 *   // with realtime updates:
 *   const { data } = useRow<'profiles'>({
 *     table: 'profiles',
 *     value: userId,
 *     realtime: true,
 *   });
 */
export function useRow<T extends TableName>(
  options: UseRowOptions<T>
): UseRowResult<Row<T>> {
  const { table, column = 'id', value, realtime = false } = options;
  const [data, setData] = useState<Row<T> | null>(null);
  const [isLoading, setIsLoading] = useState<boolean>(false);
  const [error, setError] = useState<Error | null>(null);

  useEffect(() => {
    if (value === null || value === undefined) {
      setData(null);
      setIsLoading(false);
      setError(null);
      return;
    }

    let cancelled = false;
    setIsLoading(true);
    setError(null);

    const fetchInitial = async () => {
      const { data: row, error: fetchError } = await supabase
        .from(table)
        .select('*')
        .eq(column, value)
        .maybeSingle();

      if (cancelled) return;
      if (fetchError) {
        setError(new Error(fetchError.message));
        setData(null);
      } else {
        setData(row as Row<T> | null);
      }
      setIsLoading(false);
    };

    fetchInitial();

    if (!realtime) {
      return () => {
        cancelled = true;
      };
    }

    // Realtime subscription
    const channel = supabase
      .channel(`${table}:${column}:${value}`)
      .on(
        'postgres_changes',
        {
          event: '*',
          schema: 'public',
          table: table as string,
          filter: `${column}=eq.${value}`,
        },
        (payload) => {
          if (cancelled) return;
          if (payload.eventType === 'DELETE') {
            setData(null);
          } else {
            setData(payload.new as Row<T>);
          }
        }
      )
      .subscribe();

    return () => {
      cancelled = true;
      supabase.removeChannel(channel);
    };
  }, [table, column, value, realtime]);

  return { data, isLoading, error };
}
```

```ts
}

/**
 * React hook to fetch a list of rows from a table, with optional filtering/
 * ordering and optional realtime updates. Drop-in replacement for Firebase
 * useCollection().
 *
 * Usage:
 *   const { data: posts } = useCollection<'posts'>({
 *     table: 'posts',
 *     query: (q) => q.eq('user_id', userId).order('created_at', { ascending: false }),
 *     realtime: { column: 'user_id', value: userId },
 *     deps: [userId],
 *   });
 */
export function useCollection<T extends TableName>(
  options: UseCollectionOptions<T>
): UseCollectionResult<Row<T>> {
  const { table, query, realtime, deps } = options;
  const [data, setData] = useState<Row<T>[] | null>(null);
  const [isLoading, setIsLoading] = useState<boolean>(false);
  const [error, setError] = useState<Error | null>(null);

  useEffect(() => {
    let cancelled = false;

    const run = async () => {
      setIsLoading(true);
      setError(null);

      const base = baseQuery(table);
      const finalQuery = (query ? query(base) : base) as unknown as PostgrestFilterBuilder<
        Database['public'],
        Row<T>,
        Row<T>[],
        string
      >;

      const { data: rows, error: queryError } = await finalQuery;

      if (cancelled) return;
      if (queryError) {
        setError(new Error(queryError.message));
        setData(null);
      } else {
        setData((rows ?? []) as Row<T>[]);
      }
      setIsLoading(false);
    };

    run();

    if (!realtime) {
      return () => {
        cancelled = true;
      };
    }

    // Realtime: refetch on any matching change
    const channel = supabase
      .channel(`${table}:${realtime.column}:${realtime.value}`)
      .on(
        'postgres_changes',
        {
          event: '*',
          schema: 'public',
          table: table as string,
          filter: `${realtime.column}=eq.${realtime.value}`,
        },
        () => {
          if (!cancelled) run();
        }
      )
      .subscribe();

    return () => {
      cancelled = true;
      supabase.removeChannel(channel);
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, deps);

  return { data, isLoading, error };
}
```

```ts
'use client';

import { useState, useEffect } from 'react';
import { useRouter, usePathname } from 'next/navigation';
import { supabase } from '@/supabase/client';
import { useSupabaseAuth } from '@/supabase/provider';
import type { Tables } from '@/supabase/types';

type Profile = Tables<'profiles'>;

const PROFILE_CACHE_KEY = (uid: string) => `profile_cache_${uid}`;

/**
 * useAuth — Supabase version. Reads the auth user from the SupabaseProvider,
 * fetches the profiles row from Postgres, gates on email verification, and
 * redirects unauthenticated users as before.
 *
 * Return shape matches the old Firebase hook so call sites don't break:
 *   { user, profile, loading }
 */
export function useAuth() {
  const { user, isUserLoading: authLoading } = useSupabaseAuth();
  const router = useRouter();
  const pathname = usePathname();
  const [profile, setProfile] = useState<Profile | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    if (authLoading) return;

    const publicPaths = ['/', '/auth/login', '/auth/signup'];
    const isPublicPath = publicPaths.includes(pathname ?? '');

    // 1. Unauthenticated
    if (!user) {
      setProfile(null);
      setLoading(false);
      if (!isPublicPath && pathname !== '/auth/verify') {
        router.push('/auth/login');
      }
      return;
    }

    // 2. Email verification gate — Supabase sets email_confirmed_at
    //    once the user clicks the confirmation link.
    const emailVerified = Boolean(user.email_confirmed_at);
    if (!emailVerified) {
      setProfile(null);
      setLoading(false);
      if (!isPublicPath && pathname !== '/auth/verify') {
        router.push('/auth/verify');
      }
      return;
    }

    // 3. Load cached profile (except on /profile where we want fresh data)
    const isProfilePage = pathname === '/profile';
    if (!isProfilePage) {
      try {
        const cached = localStorage.getItem(PROFILE_CACHE_KEY(user.id));
        if (cached) {
          setProfile(JSON.parse(cached) as Profile);
          setLoading(false);
        } else {
          setLoading(true);
        }
      } catch {
        setLoading(true);
      }
    } else {
      setLoading(true);
    }

    // 4. Fetch fresh profile from Postgres
    let cancelled = false;
    supabase
      .from('profiles')
      .select('*')
      .eq('id', user.id)
      .maybeSingle()
      .then(({ data, error }) => {
        if (cancelled) return;
        if (error) {
          console.error('[useAuth] failed to load profile', error);
          setLoading(false);
          return;
        }
        if (data) {
          setProfile(data);
          try {
            localStorage.setItem(PROFILE_CACHE_KEY(user.id), JSON.stringify(data));
          } catch {}
        }
        setLoading(false);
      });

    return () => {
      cancelled = true;
    };
  }, [user, authLoading, pathname, router]);

  return { user, profile, loading: authLoading || loading };
}
```

```tsx
'use client';

import { useEffect, useState } from 'react';
import { useRouter, useSearchParams } from 'next/navigation';
import { supabase } from '@/supabase/client';
import { useSupabaseAuth } from '@/supabase/provider';
import { Button } from '@/components/ui/button';
import { useToast } from '@/hooks/use-toast';

export default function VerifyEmailPage() {
  const router = useRouter();
  const params = useSearchParams();
  const { user } = useSupabaseAuth();
  const { toast } = useToast();
  const [status, setStatus] = useState<'idle' | 'verifying' | 'done' | 'error'>('idle');
  const [resending, setResending] = useState(false);

  // Handle click-through from verification email
  useEffect(() => {
    const tokenHash = params?.get('token_hash');
    const type = params?.get('type');
    if (!tokenHash || !type) return;

    setStatus('verifying');
    supabase.auth
      .verifyOtp({ token_hash: tokenHash, type: type as 'email' })
      .then(({ error }) => {
        if (error) {
          setStatus('error');
          toast({
            variant: 'destructive',
            title: 'Verification failed',
            description: error.message,
          });
          return;
        }
        setStatus('done');
        toast({ title: 'Email verified!', description: 'You can start using TrueVibe.' });
        router.push('/matches');
      });
  }, [params, router, toast]);

  // If already verified, redirect
  useEffect(() => {
    if (user?.email_confirmed_at) {
      router.push('/matches');
    }
  }, [user, router]);

  const handleResend = async () => {
    if (!user?.email) return;
    setResending(true);
    const { error } = await supabase.auth.resend({
      type: 'signup',
      email: user.email,
      options: {
        emailRedirectTo: `${window.location.origin}/auth/verify`,
      },
    });
    setResending(false);
    if (error) {
      toast({
        variant: 'destructive',
        title: 'Could not resend',
        description: error.message,
      });
    } else {
      toast({ title: 'Verification email sent', description: 'Check your inbox.' });
    }
  };

  return (
    <div className="min-h-screen flex items-center justify-center p-6">
      <div className="max-w-md text-center space-y-4">
        <h1 className="text-2xl font-bold">Verify your email</h1>
        {status === 'verifying' && <p>Verifying your email...</p>}
        {status === 'error' && <p className="text-red-500">Verification failed. Try the link again or resend.</p>}
        {status === 'idle' && (
          <>
            <p className="text-muted-foreground">
              We sent a verification link to <b>{user?.email ?? 'your email'}</b>. Click it to finish signing up.
            </p>
            <Button onClick={handleResend} disabled={resending}>
              {resending ? 'Sending...' : 'Resend verification email'}
            </Button>
          </>
        )}
      </div>
    </div>
  );
}
```

### Task 5.2: Map the Firebase → Supabase conversion pattern

**Files:** none (reference)

The conversion from Firebase SDK calls to Supabase is mechanical. Use this cheat sheet while rewriting feature pages in Phases 5-8:

| Firebase pattern | Supabase equivalent |
|---|---|
| `doc(db, 'users', uid)` → `getDoc(ref)` | `supabase.from('profiles').select('*').eq('id', uid).maybeSingle()` |
| `setDoc(ref, data, { merge: true })` | `supabase.from('profiles').update(data).eq('id', uid)` |
| `addDoc(collection(db, 'posts'), data)` | `supabase.from('posts').insert(data).select().single()` |
| `updateDoc(ref, patch)` | `supabase.from(TABLE).update(patch).eq('id', rowId)` |
| `deleteDoc(ref)` | `supabase.from(TABLE).delete().eq('id', rowId)` |
| `query(collection, where('x','==',v), orderBy('y','desc'))` | `.select('*').eq('x', v).order('y', { ascending: false })` |
| `onSnapshot(query, cb)` | `useCollection({ table, query, realtime, deps })` (see Phase 6) |
| `useCollection(q)` from @/firebase | `useCollection` from @/supabase/hooks/use-collection |
| `useDoc(ref)` from @/firebase | `useRow({ table, value })` from @/supabase/hooks/use-row |
| `storage.ref('path').uploadBytes(file)` | `supabase.storage.from(BUCKET).upload(path, file)` |
| `getDownloadURL(ref)` | `supabase.storage.from(BUCKET).getPublicUrl(path).data.publicUrl` |
| `non-blocking setDoc + error-emitter` | `await supabase.from(...).update(...)` + toast on error |

```tsx
      )
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, [user, toast]);

  return null;
```

New flow (per Section 1 decision #4 — no ID photo storage):

```
Browser (/verification page):
  User picks a file from <input type="file" />
  File is read into a base64 data URL (or sent directly as multipart/form-data)
  Client POSTs to /api/verify-identity with the file
  Client DISCARDS the file from React state after the response

  2. Parse the multipart body to extract the image

  4. Flow uses Gemini to judge the photo
  5. On approve: supabaseAdmin.from('profiles').update({
       identity_verified: true,
       verification_status: 'verified',
     }).eq('id', userId)
  6. On reject: supabaseAdmin.from('profiles').update({
       verification_status: 'rejected',
     }).eq('id', userId)
  7. Return the decision to the client
  8. IMAGE IS NEVER WRITTEN TO STORAGE
```

Sketch of `route.ts`:

```ts
import { NextResponse } from 'next/server';
import { verifyAuth } from '@/lib/verify-auth';
import { supabaseAdmin } from '@/lib/supabase-admin';
import { verifyIdentityFlow } from '@/ai/flows/verify-identity-flow';

export async function POST(request: Request) {
  try {
    const user = await verifyAuth();
    const formData = await request.formData();
    const file = formData.get('image') as File | null;
    if (!file) {
      return NextResponse.json({ error: 'No image provided' }, { status: 400 });
    }
    const arrayBuffer = await file.arrayBuffer();
    const base64 = Buffer.from(arrayBuffer).toString('base64');

    const result = await verifyIdentityFlow({
      imageBase64: base64,
      mimeType: file.type,
    });

    if (result.approved) {
      await supabaseAdmin
        .from('profiles')
        .update({
          identity_verified: true,
          verification_status: 'verified',
        })
        .eq('id', user.id);
    } else {
      await supabaseAdmin
        .from('profiles')
        .update({ verification_status: 'rejected' })
        .eq('id', user.id);
    }

    return NextResponse.json(result);
  } catch (err: any) {
    return NextResponse.json({ error: err.message }, { status: 401 });
  }
}
```
